package game

import (
    "encoding/json"
    "os"
    "path/filepath"
    "reflect"
    "testing"
)

// Tool-only validation of a real GUI save/reopen trace. This is a remake
// clock-continuation check; it is not an original-oracle experiment.
func TestValidateGUICombatReentryClock(t *testing.T) {
    dir := os.Getenv("SUNDOG_GUI_REENTRY")
    if dir == "" { t.Skip("GUI trace not supplied") }
    data, err := os.ReadFile(filepath.Join(dir, "reentry-loaded-save.json"))
    if err != nil { t.Fatal(err) }
    var saved struct { State State `json:"state"` }
    if err = json.Unmarshal(data, &saved); err != nil { t.Fatal(err) }
    data, err = os.ReadFile(filepath.Join(dir, "reentry-observed.json"))
    if err != nil { t.Fatal(err) }
    var observed State
    if err = json.Unmarshal(data, &observed); err != nil { t.Fatal(err) }
    delta := int64(observed.AwakeClock.Source.Ticks)-int64(saved.State.AwakeClock.Source.Ticks)
    if delta < 0 || delta > 300 { t.Fatalf("unbounded clock continuation: %d", delta) }
    expected := saved.State
    expected.QuitRequested, expected.Saved = false, false
    r, err := NewReducer()
    if err != nil { t.Fatal(err) }
    expected, err = r.Step(expected, ActionOpenTactical)
    if err != nil { t.Fatal(err) }
    for range int(delta) { advanceAwakeVBL(&expected) }
    // Ignore transient messages and overlay screens, but retain every combat
    // field the GUI driver previously compared, plus cargo and saved clocks.
    keys := []string{"fuel","enemyType","enemyHull","enemyMaxHull","combatRange",
        "enemyBearing","enemyPitch","weaponCharge","combatRolls","enemyFireCountdown",
        "enemyDriftBearing","enemyDriftPitch","hullCondition","shieldLevel","shieldSetting",
        "shieldsActive","shipSystems","selectedSystemIndex","shipPhase","holds",
        "timeMinutes","awakeClock","vigor","rest","health","nourishment"}
    asMap := func(s State) map[string]any {
        b, e := json.Marshal(s); if e != nil { t.Fatal(e) }
        var v map[string]any; if e = json.Unmarshal(b, &v); e != nil { t.Fatal(e) }; return v
    }
    want, got := asMap(expected), asMap(observed)
    for _, key := range keys {
        if !reflect.DeepEqual(want[key], got[key]) { t.Errorf("%s after %d VBL: want %v got %v", key, delta, want[key], got[key]) }
    }
    t.Logf("validated %d fields after %d saved-clock VBL inputs", len(keys), delta)
}
