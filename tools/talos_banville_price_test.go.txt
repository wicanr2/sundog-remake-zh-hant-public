// 正常普通城市交易的數值與record比較；原版選單／RND刷新另驗。
func TestSunDogNormalMarketSale(t *testing.T) {
	paths := []string{os.Getenv("SUNDOG_NORMAL_TRADE_INPUT"), os.Getenv("SUNDOG_NORMAL_TRADE_AFTER"), os.Getenv("SUNDOG_NORMAL_TRADE_SETTLED")}
	if paths[0] == "" || paths[1] == "" || paths[2] == "" {
		t.Skip("需要正常確認、批准及離開交易所三份保存")
	}
	type lot struct{ Goods, Quality, Price, SourceCode int }
	type market struct{ Supply, Demand []int }
	type state struct {
		Schema                                               int
		Screen, CityID, SystemID, PlanetID, GroundMode       string
		TradeSaleSlot, TradeSalePrice, TradePage, Cash, Bank int
		Holds                                                []lot
		Warehouses                                           map[string][]lot
		Markets                                              map[string]market
		TradeSession                                         struct {
			Active                          bool
			InitialCash, InitialBank, Funds int
		}
	}
	read := func(path string) state {
		raw, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		var e struct{ Slots map[string]state }
		if err = json.Unmarshal(raw, &e); err != nil {
			t.Fatal(err)
		}
		return e.Slots["slot-1"]
	}
	in, after, settled := read(paths[0]), read(paths[1]), read(paths[2])
	schema, err := strconv.Atoi(os.Getenv("SUNDOG_NORMAL_TRADE_SCHEMA"))
	if err != nil || schema < 44 || in.Schema != schema || after.Schema != schema || settled.Schema != schema {
		t.Fatal("僅接受目前schema正常保存，不遷移")
	}
	if in.Screen != "trade" || in.CityID == "banville" || in.GroundMode != "pod" || in.TradePage != 9 || in.TradeSaleSlot < 5 || in.TradeSaleSlot > 6 || len(in.Holds) != 2 || !in.TradeSession.Active {
		t.Fatal("不是正常普通城市pod確認")
	}
	if after.CityID != in.CityID || after.SystemID != in.SystemID || after.Screen != "trade" || after.TradePage != 0 || settled.CityID != in.CityID || settled.Screen != "ground" || settled.TradeSession.Active {
		t.Fatal("批准／結算保存來源不一致")
	}
	type record struct {
		Index int
		Name  string
	}
	var world struct {
		Tables map[string]struct{ Records []record }
	}
	raw, err := os.ReadFile(os.Getenv("SUNDOG_NORMAL_TRADE_WORLD"))
	if err != nil {
		t.Fatal(err)
	}
	if err = json.Unmarshal(raw, &world); err != nil {
		t.Fatal(err)
	}
	slug := func(name string) string {
		return strings.Trim(regexp.MustCompile("[^a-z0-9]+").ReplaceAllString(strings.ToLower(name), "-"), "-")
	}
	index := func(table, id string) int {
		for _, r := range world.Tables[table].Records {
			if slug(r.Name) == id {
				return r.Index
			}
		}
		t.Fatalf("找不到來源%s／%s", table, id)
		return -1
	}
	system, planet, city := index("systems", in.SystemID), index("planets", in.PlanetID), index("cities", in.CityID)+1
	inputLot := in.Holds[in.TradeSaleSlot-5]
	if inputLot.Goods < 2 || inputLot.Goods > 30 || inputLot.Quality < 0 || inputLot.Quality > 7 || inputLot.SourceCode < 1 || inputLot.SourceCode > 60 {
		t.Fatal("正常貨物範圍不符")
	}
	m := in.Markets[in.CityID]
	if len(m.Supply) != 32 || len(m.Demand) != 32 {
		t.Fatal("正常供需矩陣長度不符")
	}
	disk := sundogWalkerDisk(t)
	h := banvilleHarness(t, disk)
	for i, b := range disk[0x5600:0x5800] {
		h.mem[depotLocal(0x46000, 3)+uint32(i)] = b
	}
	for i, b := range disk[0x5800:0x58f0] {
		h.mem[cadenceGlobal(0x11f)+uint32(i)] = b
	}
	for i, b := range disk[0x58f0:0x5a10] {
		h.mem[cadenceGlobal(0x197)+uint32(i)] = b
	}
	for i, b := range disk[0x6a00:0x6c00] {
		h.mem[cadenceOuter(0xa5)+uint32(i)] = b
	}
	h.poke(cadenceGlobal(0x2a), uint16(system+1|(planet+1)<<4|city<<9))
	h.poke(cadenceOuter(0xd), 0)
	systemPtr := cadenceGlobal(0x11f) + uint32(system*20)
	word9 := h.peek(systemPtr + 18)
	h.poke(cadenceOuter(7), word9&15)
	h.poke(cadenceOuter(6), (word9>>4)&15)
	for local, values := range map[int][]int{0x1b: m.Demand, 0x22: m.Supply, 0x32: m.Demand, 0x2b: m.Supply} {
		for goods := 1; goods < 32; goods++ {
			if values[goods] < 0 || values[goods] > 7 {
				t.Fatal("供需超出原始3bit範圍")
			}
			ptr := cadenceOuter(local) + uint32((goods-1)/5*2)
			h.poke(ptr, h.peek(ptr)|uint16(values[goods])<<uint((goods-1)%5*3))
		}
	}
	pack := func(l lot) [2]uint16 {
		return [2]uint16{uint16(l.Goods | l.Quality<<5 | l.SourceCode<<8), uint16(l.Price)}
	}
	packed := pack(inputLot)
	h.poke(0x47000, packed[0])
	h.poke(0x47002, packed[1])
	h.poke(depotLocal(0x46000, 10), 0x7000)
	h.poke(depotLocal(0x46000, 9), 1)
	banvilleRun(h, disk, 0x44200, 0x922, 0xa3e)
	price := int(int16(h.peek(depotLocal(0x46000, 11))))
	// 收入金額採既有價格word×100契約；本切片直接驗原始收款caller，未重跑確認頁金額轉換。
	income := price * 100
	bankPtr := systemPtr + 12
	depotPutMoney(h, bankPtr, in.TradeSession.InitialBank)
	depotPutMoney(h, cadenceGlobal(0x33), in.TradeSession.InitialCash)
	h.poke(depotLocal(0x46000, 0x103), uint16(systemPtr-0x40000))
	banvilleRun(h, disk, 0x44200, 0x3fb, 0x404)
	banvilleRun(h, disk, 0x44200, 0x447, 0x469)
	fundsBefore := depotMoney(h, cadenceOuter(0x13))
	if fundsBefore != in.TradeSession.Funds {
		t.Fatal("本入口限單筆普通售出session，初態資金不同")
	}
	depotPutMoney(h, depotLocal(0x46000, 3), income)
	banvilleRun(h, disk, 0x44200, 0x11f4, 0x11fa)
	fundsAfter := depotMoney(h, cadenceOuter(0x13))
	banvilleRun(h, disk, 0x44200, 0x4b6, 0x51a)
	cashAfter, bankAfter := depotMoney(h, cadenceGlobal(0x33)), depotMoney(h, bankPtr)
	ptrs := [6]uint32{cadenceOuter(0x39), cadenceOuter(0x39) + 4, cadenceOuter(0x39) + 8, cadenceOuter(0x39) + 12, cadenceGlobal(0xec), cadenceGlobal(0xec) + 4}
	if len(in.Warehouses[in.CityID]) != 4 || len(after.Warehouses[in.CityID]) != 4 || len(after.Holds) != 2 {
		t.Fatal("貨物record數不符")
	}
	var slotsBefore, slotsAfter, wantedSlots [6][2]uint16
	for i := 0; i < 6; i++ {
		if i < 4 {
			slotsBefore[i] = pack(in.Warehouses[in.CityID][i])
			wantedSlots[i] = pack(after.Warehouses[in.CityID][i])
		} else {
			slotsBefore[i] = pack(in.Holds[i-4])
			wantedSlots[i] = pack(after.Holds[i-4])
		}
		h.poke(ptrs[i], slotsBefore[i][0])
		h.poke(ptrs[i]+2, slotsBefore[i][1])
	}
	h.poke(depotLocal(0x46000, 2), uint16(in.TradeSaleSlot))
	h.poke(depotLocal(0x46000, 5), packed[0])
	h.poke(depotLocal(0x46000, 6), packed[1])
	stream := uint32(operandStream + 0x8000)
	h.startDispatch(disk[0x453fa:0x4542a], stream)
	calls, finished := 0, false
	for steps := 0; steps < 10000; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp := h.cpu.State.A[4], h.cpu.State.SSP
			at := int(ip-stream) + 0x11fa
			if at == 0x1226 {
				finished = true
				break
			}
			if h.mem[ip] == 0x70 && h.mem[ip+1] == 0x15 {
				p := depotLocal(0x46000, 5)
				if at != 0x11fe || calls != 0 || h.peek(sp) != 0 || h.peek(sp+2) != 4 || h.peek(sp+4) != 0 || h.peek(sp+6) != uint16(p-0x40000) {
					t.Fatal("售出FILLCHAR ABI不符")
				}
				for b := uint32(0); b < 4; b++ {
					h.mem[p+b] = 0
				}
				calls++
				h.cpu.State.SSP = sp + 8
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			if h.mem[ip] == 0x70 || h.mem[ip] == 0x73 || h.mem[ip] == 0x90 || h.mem[ip] == 0xef {
				t.Fatal("未知清槽呼叫")
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			t.Fatal(err)
		}
	}
	if !finished || calls != 1 {
		t.Fatal("原始清槽未完成")
	}
	for i, p := range ptrs {
		slotsAfter[i] = [2]uint16{h.peek(p), h.peek(p + 2)}
	}
	differences := 0
	for _, same := range []bool{price == in.TradeSalePrice, fundsAfter == after.TradeSession.Funds, cashAfter == settled.Cash, bankAfter == settled.Bank, slotsAfter[4] == wantedSlots[4] && slotsAfter[5] == wantedSlots[5], slotsAfter[0] == wantedSlots[0] && slotsAfter[1] == wantedSlots[1] && slotsAfter[2] == wantedSlots[2] && slotsAfter[3] == wantedSlots[3]} {
		if !same {
			differences++
		}
	}
	report := struct {
		Scope, Boundaries                                                                                                                string
		Input                                                                                                                            state
		SourceSystemIndex, SourcePlanetIndex, SourceCityIndex, Price, Income, FundsBefore, FundsAfter, CashAfter, BankAfter, Differences int
		SlotsBefore, SlotsAfter                                                                                                          [6][2]uint16
	}{"正常普通城市確認／批准／結算保存；XDOTRADI raw0x44200:0922..0a3d完整估值、03fb..0403／0447..0468載入資金、11f4..11f9一般收款、04b6..0519完整結算、11fa..1225清槽；六項數值及record比較。", "供需與來源碼取實際GUI保存；原版價格／金額helper及清槽分支真正執行。FILLCHAR為嚴格標準ABI。收入沿既有價格word×100契約，未重跑完整批准頁轉換；不執行報價刷新、普通市場RND減量或原版GUI，不宣稱自然骰序／整段交易exact。", in, system, planet, city, price, income, fundsBefore, fundsAfter, cashAfter, bankAfter, differences, slotsBefore, slotsAfter}
	tradeClassExport(t, "native-normal-market-sale.json", report)
	if differences != 0 {
		t.Fatalf("正常市場六項不同 %+v", report)
	}
	t.Logf("PASS:正常%s貨物%d估值%d收入%d／Funds%d／Cash%d／Bank%d，六項零差異", in.CityID, inputLot.Goods, price, income, fundsAfter, cashAfter, bankAfter)
}

// 正常玩家保存只作oracle受控初態；remake不讀原版RAM。
// docs/atari-st/banville-restoration-spec.md的正常第一期／冷凍人玩家鏈。
func TestSunDogBanvilleNormalDelivery(t *testing.T) {
	inputPath, afterPath := os.Getenv("SUNDOG_NORMAL_DELIVERY_INPUT"), os.Getenv("SUNDOG_NORMAL_DELIVERY_AFTER")
	if inputPath == "" || afterPath == "" {
		t.Skip("需要正常GUI確認及交付後保存")
	}
	type lot struct{ Goods, Quality, Price, SourceCode int }
	type colony struct {
		Initialized                   bool
		Phase, NextPhase, Accumulated int
		Remaining                     [32]int
	}
	type state struct {
		Schema                                                                   int
		Screen, CityID, SystemID, GroundMode                                     string
		TradeClassShift, TradeSaleSlot, TradeSalePrice, TradePage, BanvilleCargo int
		Cash, Bank                                                               int
		MissionComplete                                                          bool
		OutcomeKey                                                               string
		Holds                                                                    []lot
		Colony                                                                   colony
		TradeSession                                                             struct{ Funds int }
	}
	read := func(path string) state {
		raw, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		var envelope struct{ Slots map[string]state }
		// Final Score uses the actual Q autosave envelope; it cannot enter
		// the F5 library. Both forms are observations, never runtime inputs.
		var single struct{ State state }
		if err := json.Unmarshal(raw, &envelope); err != nil {
			t.Fatal(err)
		}
		if len(envelope.Slots)==0 {
			if err:=json.Unmarshal(raw,&single);err!=nil {t.Fatal(err)}
			if single.State.Schema==0 {t.Fatal("沒有正式保存狀態")}
			return single.State
		}
		return envelope.Slots["slot-1"]
	}
	in, after := read(inputPath), read(afterPath)
	currentSchema, err := strconv.Atoi(os.Getenv("SUNDOG_NORMAL_DELIVERY_SCHEMA"))
	if err != nil || currentSchema < 44 {
		t.Fatal("研究入口須明示目前schema；舊格式只作歷史回驗，不提供玩家遷移")
	}
	if (in.Schema != currentSchema && in.Schema != 40 && in.Schema != 41) || in.Screen != "trade" || in.CityID != "banville" || in.SystemID != "jondd" || in.GroundMode != "pod" || !in.Colony.Initialized || in.Colony.Phase < 0 || in.Colony.Phase > 8 || in.Colony.NextPhase != in.Colony.Phase+1 || in.TradeSaleSlot < 5 || in.TradeSaleSlot > 6 || len(in.Holds) != 2 || in.TradePage != 9 || in.TradeSalePrice < 0 || in.TradeClassShift < 0 || in.TradeClassShift > 17 {
		t.Fatal("不是正常需求pod確認保存")
	}
	final := after.Colony.Phase == 9 && after.MissionComplete && after.Screen == "score" && after.OutcomeKey == "ui.outcome.success"
	if after.Schema != in.Schema || (!final && after.Screen != in.Screen) || after.CityID != in.CityID || after.SystemID != in.SystemID || after.GroundMode != in.GroundMode || after.TradePage != 0 {
		t.Fatal("交付後保存與確認來源／schema或頁面不符")
	}
	lotInput := in.Holds[in.TradeSaleSlot-5]
	if lotInput.Goods < 1 || lotInput.Goods > 31 || in.Colony.Remaining[lotInput.Goods] <= 0 {
		t.Fatal("確認貨物沒有當期需求")
	}
	disk := sundogWalkerDisk(t)
	h := banvilleHarness(t, disk)
	for i, b := range disk[0x5600:0x5800] {
		h.mem[depotLocal(0x46000, 3)+uint32(i)] = b
	}
	for i, b := range disk[0x5800:0x58f0] {
		h.mem[cadenceGlobal(0x11f)+uint32(i)] = b
	}
	for i, b := range disk[0x58f0:0x5a10] {
		h.mem[cadenceGlobal(0x197)+uint32(i)] = b
	}
	for i, b := range disk[0x6a00:0x6c00] {
		h.mem[cadenceOuter(0xa5)+uint32(i)] = b
	}
	h.poke(cadenceGlobal(0x2a), 1|1<<4|60<<9)
	h.mem[cadenceGlobal(0x10b)+3] = byte(in.TradeClassShift)
	banvilleRun(h, disk, 0x44200, 0x292, 0x31f)
	word9 := h.peek(cadenceGlobal(0x11f) + 18)
	h.poke(cadenceOuter(7), word9&15)
	h.poke(cadenceOuter(6), (word9>>4)&15)
	h.poke(cadenceOuter(0xd), 1)
	for goods := 1; goods < 32; goods++ {
		ptr := cadenceOuter(0x1b) + uint32((goods-1)/5*2)
		h.poke(ptr, h.peek(ptr)|uint16(in.Colony.Remaining[goods])<<uint((goods-1)%5*3))
	}
	h.mem[cadenceGlobal(0x10b)+1] = byte(in.Colony.Phase)
	h.mem[cadenceGlobal(0x10b)+2] = byte(in.Colony.NextPhase)
	h.poke(cadenceOuter(2), uint16(in.Colony.Accumulated))
	h.poke(cadenceGlobal(0x41), uint16(in.BanvilleCargo))
	depotPutMoney(h, cadenceOuter(0x13), in.TradeSession.Funds)
	packed := uint16(lotInput.Goods | lotInput.Quality<<5 | lotInput.SourceCode<<8)
	h.poke(0x47000, packed)
	h.poke(0x47002, uint16(lotInput.Price))
	h.poke(depotLocal(0x46000, 10), 0x7000)
	h.poke(depotLocal(0x46000, 9), 1)
	banvilleRun(h, disk, 0x44200, 0x922, 0xa3e)
	price := int(int16(h.peek(depotLocal(0x46000, 11))))
	// current_price使用caller同號local；回復實際購得lot才執行交付。
	h.poke(depotLocal(0x46000, 5), packed)
	h.poke(depotLocal(0x46000, 6), uint16(lotInput.Price))
	_, stop := banvilleRun(h, disk, 0x44200, 0x11af, 0x11e4)
	report := struct {
		Scope, Boundaries                                                              string
		Input                                                                          state
		Lot                                                                            lot
		Price, AccumulatorAfter, FundsAfter, ScoreAfter, CurrentAfter, NextAfter, Stop int
		Remaining                                                                      [32]int
		Differences                                                                    int
	}{Scope: "明示目前schema正常玩家確認保存；40／41僅歷史研究。比較同貨物／分類位移／來源／購入價／需求／累積／資金／score；XDOTRADI raw0x44200:0292..031e原始分類、0922..0a3d完整估值、11af..11e4確認後交付", Boundaries: "SCAN／圖形／聲音／子程序返回沿既有明示邊界；不執行原版GUI批准、quote refresh或自然RND。同星系／行星來源估值不讀保存城址。普通交付清槽由既有六槽writer及正常GUI支持；最終期在0x0118走EXIT邊界，早於0x11fa..0x1225清槽，不外推原版EXIT後記憶體", Input: in, Lot: lotInput, Price: price, AccumulatorAfter: int(h.peek(cadenceOuter(2))), FundsAfter: depotMoney(h, cadenceOuter(0x13)), ScoreAfter: int(h.peek(cadenceGlobal(0x41))), CurrentAfter: int(h.mem[cadenceGlobal(0x10b)+1]), NextAfter: int(h.mem[cadenceGlobal(0x10b)+2]), Stop: stop, Remaining: banvilleDemand(h)}
	if price != in.TradeSalePrice {
		report.Differences++
	}
	if report.AccumulatorAfter != after.Colony.Accumulated {
		report.Differences++
	}
	guiFunds := after.TradeSession.Funds
	if final {
		// Score closes the remake trade session. Compare its settled wallet
		// projection to the original caller's unchanged funds; this does not
		// claim a new native settlement or original score GUI experiment.
		guiFunds = after.Cash + max(0, after.Bank)
		if stop != 0x11e2 {
			t.Fatal("原版未走最終計分呼叫", stop)
		}
	}
	if report.FundsAfter != guiFunds {
		report.Differences++
	}
	if report.ScoreAfter != after.BanvilleCargo {
		report.Differences++
	}
	if report.CurrentAfter != after.Colony.Phase || report.NextAfter != after.Colony.NextPhase {
		report.Differences++
	}
	if report.Remaining != after.Colony.Remaining {
		report.Differences++
	}
	tradeClassExport(t, "native-banville-normal-delivery.json", report)
	if report.Differences != 0 {
		t.Fatalf("正常交付與原版不同 %+v", report)
	}
	t.Logf("PASS:實際Goods%d Quality%d Source%d Price%d -> valuation%d accumulated%d funds%d，六項原生／GUI結果零差異", lotInput.Goods, lotInput.Quality, lotInput.SourceCode, lotInput.Price, price, report.AccumulatorAfter, report.FundsAfter)
}

// 原始pick_item寫入流程，選單／圖形與輸入是嚴格ABI邊界。
// 第一空槽helper及最終MOV仍實際執行；沒有代換其分派答案。
func purchasePicker(h *harness, disk []byte, choice, depth int) (menus, inputs, helpers int) {
	frame := h.cpu.State.A[0]
	stream := uint32(operandStream + 0x8000 + depth*0x1000)
	h.startDispatch(disk[0x45208:0x4538a], stream)
	for step := 0; step < 100000; step++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp := h.cpu.State.A[4], h.cpu.State.SSP
			at := int(ip-stream) + 0x1008
			if at == 0x1186 {
				return
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			pop, handled := 0, false
			switch {
			case op == 0xef && proc == 0xb && at == 0x1028:
				if h.peek(sp+12) != 2 || h.peek(sp+14) != 1 || h.peek(sp+16) != 14 {
					h.t.Fatal("目的地menu14 ABI")
				}
				menus++
				pop = 9
				handled = true
			case op == 0xef && proc == 2 && (at == 0x102c || at == 0x1078):
				if h.peek(sp) != 5 || h.peek(sp+2) != 0 {
					h.t.Fatal("pick_item圖形reset ABI")
				}
				pop = 2
				handled = true
			case op == 0xef && proc == 3 && at == 0x1059:
				if h.peek(sp) != 0 {
					h.t.Fatal("目的地輸入return slot")
				}
				h.poke(sp, uint16(choice))
				inputs++
				handled = true
			case op == 0xef && proc == 0xe && at == 0x1080:
				pop = 2
				handled = true // 六槽畫法另驗，不製造原版framebuffer。
			case op == 0xef && proc == 0xd && (at == 0x10c2 || at == 0x10f1):
				child := uint32(0x4a000 + depth*0x100)
				h.poke(child, 0x5000)
				h.poke(depotLocal(child, 2), h.peek(sp))
				h.poke(depotLocal(child, 3), h.peek(sp+2))
				levelsFrame(h, child)
				_, stop := banvilleRunDepth(h, disk, 0x44200, 0x58e, 0x5e6, depth+1)
				if stop != 0x5e6 {
					h.t.Fatal("目的地第一空槽未停止")
				}
				levelsFrame(h, frame)
				helpers++
				pop = 2
				handled = true
			case op == 0x73 && proc == 0x16:
				handled = true
			case op == 0x73 && proc == 0x1b:
				pop = 2
				handled = true
			case op == 0x73 && proc == 0x2c && at == 0x110f:
				if h.peek(sp) != 120 {
					h.t.Fatal("放貨目的地提示參數")
				}
				pop = 1
				handled = true
			case op == 0x70 || op == 0x72 || op == 0x73 || op == 0x90 || op == 0xef:
				h.t.Fatalf("未核定pick_item呼叫%x op%x proc%x", at, op, proc)
			}
			if handled {
				h.cpu.State.SSP = sp + uint32(pop*2)
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("目的地IP%x: %v", int(h.cpu.State.A[4]-stream)+0x1008, err)
		}
	}
	h.t.Fatal("目的地未停止")
	return
}

			}

				row.After[i] = [2]uint16{h.peek(p), h.peek(p + 2)}

				}

				}
			}

	}

	}{Scope: "XDOTRADI raw基底0x44200，售出後11fa..1225：FILLCHAR標準byte清零採嚴格四參數邊界，原版slot分支／原始兩word MOV直接執行。selector1..4倉庫、5..6 pod。初態三種pattern，非完整成交選單，也不證明獨立存倉source清空。"}
	for slot := 1; slot <= 6; slot++ {
		for _, pattern := range [][2]uint16{{0xffff, 0xffff}, {0xffb9, 32767}, {0x0121, 0}} {
			h := banvilleHarness(t, disk)
			ptrs := [6]uint32{cadenceOuter(0x39), cadenceOuter(0x39) + 4, cadenceOuter(0x39) + 8, cadenceOuter(0x39) + 12, cadenceGlobal(0xec), cadenceGlobal(0xec) + 4}
			r := row{Slot: slot}
			for i, p := range ptrs {
				h.poke(p, pattern[0])
				h.poke(p+2, pattern[1])
				r.Before[i] = pattern
			}
			h.poke(depotLocal(0x46000, 2), uint16(slot))
			h.poke(depotLocal(0x46000, 5), pattern[0])
			h.poke(depotLocal(0x46000, 6), pattern[1])
			stream := uint32(operandStream + 0x8000)
			h.startDispatch(disk[0x453fa:0x4542a], stream)
			calls := 0
			finished := false
			for steps := 0; steps < 10000; steps++ {
				if h.cpu.State.PC == interpBase+interpLoop+4 {
					ip, sp := h.cpu.State.A[4], h.cpu.State.SSP
					at := int(ip-stream) + 0x11fa
					if at == 0x1226 {
						finished = true
						break
					}
					if h.mem[ip] == 0x70 && h.mem[ip+1] == 0x15 {
						p := depotLocal(0x46000, 5)
						if at != 0x11fe || calls != 0 || h.peek(sp) != 0 || h.peek(sp+2) != 4 || h.peek(sp+4) != 0 || h.peek(sp+6) != uint16(p-0x40000) {
							t.Fatal("售出FILLCHAR ABI不符")
						}
						for b := uint32(0); b < 4; b++ {
							h.mem[p+b] = 0
						}
						calls++
						h.cpu.State.SSP = sp + 8
						h.cpu.State.A[4] = ip + 2
						h.start(interpLoop)
						continue
					}
					if h.mem[ip] == 0x70 || h.mem[ip] == 0x73 || h.mem[ip] == 0x90 || h.mem[ip] == 0xef {
						t.Fatal("未知清空呼叫")
					}
				}
				if _, err := h.cpu.Step(); err != nil {
					t.Fatal(err)
				}
			}
			if !finished || calls != 1 {
				t.Fatal("售出清空未完成")
			}
			for i, p := range ptrs {
				r.After[i] = [2]uint16{h.peek(p), h.peek(p + 2)}
				want := r.Before[i]
				if i == slot-1 {
					want = [2]uint16{}
				}
				if r.After[i] != want {
					t.Fatalf("clear slot%d changed slot%d", slot, i+1)
				}
			}
			report.Native = append(report.Native, r)
		}
	}
	tradeClassExport(t, "native-trade-slot-clear.json", report)
}

		for i, b := range disk[0x5800:0x5a10] {
			h.mem[cadenceGlobal(0x11f)+uint32(i)] = b
		}
		// 以原始 global 基址分別載入；兩表相鄰，重疊區 bytes 相同。
		for i, b := range disk[0x58f0:0x5a10] {
			h.mem[cadenceGlobal(0x197)+uint32(i)] = b
		}
		h.poke(depotLocal(0x46000, 3), uint16(city))
		banvilleRun(h, disk, 0x44200, 0x894, 0x91c)
		report.Locations = append(report.Locations, location{city, int(h.peek(cadenceOuter(4))), int(h.peek(cadenceOuter(3)))})
	}
	if output := os.Getenv("SUNDOG_PARITY_OUTPUT"); output != "" {
		data, err := json.MarshalIndent(report, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err = os.WriteFile(filepath.Join(output, "native-trade-records.json"), append(data, '\n'), 0600); err != nil {
			t.Fatal(err)
		}
	}
	t.Logf("PASS:%d original purchased/store/fetch records;256 source-city locator inputs", len(report.Records))
}

func TestSunDogTradeFunds(t *testing.T) {
	disk := sundogWalkerDisk(t)
	type row struct{ BankBefore, CashBefore, Cost, Income, AvailableBefore, AvailableAfterCost, AvailableAfterIncome, BankAfter, CashAfter int }
	report := struct {
		Scope  string
		Native []row
	}{Scope: "原始現金載入、銀行正值加入分支、實際購款扣除caller及一般賣款加入caller、完整結算；同一session先買後賣，包含負銀行借款。明示交易金額研究，不是完整報價／選單。XDOTRADI raw基底0x44200。"}
	for _, bank := range []int{-5000, -1, 0, 999, 4000} {
		for _, cash := range []int{0, 999, 1000} {
			available := cash
			if bank > 0 {
				available += bank
			}
			seen := map[int]bool{}
			for _, cost := range []int{0, available / 2, available} {
				if seen[cost] {
					continue
				}
				seen[cost] = true
				for _, income := range []int{0, 999, 5000} {
					h := banvilleHarness(t, disk)
					h.poke(cadenceGlobal(0x2a), 1)
					bp := cadenceGlobal(0x11f) + 12
					depotPutMoney(h, bp, bank)
					depotPutMoney(h, cadenceGlobal(0x33), cash)
					h.poke(depotLocal(0x46000, 0x103), uint16(cadenceGlobal(0x11f)-0x40000))
					banvilleRun(h, disk, 0x44200, 0x3fb, 0x404)
					banvilleRun(h, disk, 0x44200, 0x447, 0x469)
					r := row{BankBefore: bank, CashBefore: cash, Cost: cost, Income: income, AvailableBefore: depotMoney(h, cadenceOuter(0x13))}
					depotPutMoney(h, depotLocal(0x46000, 5), cost)
					banvilleRun(h, disk, 0x44200, 0x131e, 0x1324)
					r.AvailableAfterCost = depotMoney(h, cadenceOuter(0x13))
					depotPutMoney(h, depotLocal(0x46000, 3), income)
					banvilleRun(h, disk, 0x44200, 0x11f4, 0x11fa)
					r.AvailableAfterIncome = depotMoney(h, cadenceOuter(0x13))
					banvilleRun(h, disk, 0x44200, 0x4b6, 0x51a)
					r.BankAfter = depotMoney(h, bp)
					r.CashAfter = depotMoney(h, cadenceGlobal(0x33))
					if r.AvailableBefore != available || r.AvailableAfterCost != available-cost || r.AvailableAfterIncome != available-cost+income {
						t.Fatalf("可用資金 %+v", r)
					}
					report.Native = append(report.Native, r)
				}
			}
		}
	}
	if output := os.Getenv("SUNDOG_PARITY_OUTPUT"); output != "" {
		data, err := json.MarshalIndent(report, "", "  ")
		if err != nil {
			t.Fatal(err)
		}
		if err = os.WriteFile(filepath.Join(output, "native-trade-funds.json"), append(data, '\n'), 0600); err != nil {
			t.Fatal(err)
		}
	}
	t.Logf("PASS:%d original buy/sell financial sequences", len(report.Native))
}
