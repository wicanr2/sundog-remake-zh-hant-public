// #16；入口見 banville-restoration-spec.md 的食物與睡眠研究段。
// 原始 WINDOWLI:08/17/18 與 MAINLIB:0a 真正執行。
// 背景任務只記錄 ABI，圖形只消耗明示參數，不把局部原生執行當作 GUI 對拍。
func foodRun(h *harness, disk []byte, first, last, depth int, calls *[][3]int) {
	stream := uint32(operandStream + 0xa000 + depth*0x400)
	h.startDispatch(disk[0x31200+first:0x31200+last+4], stream)
	for steps := 0; steps < 100000; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp, parent := h.cpu.State.A[4], h.cpu.State.SSP, h.cpu.State.A[0]
			at := int(ip-stream) + first
			if at == last {
				return
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			if op == 0x90 && (proc == 0x17 || proc == 0x18) {
				frame := uint32(0x46000 + (depth+1)*0x200)
				entry, end, argc := 0x230, 0x2c6, 5
				if proc == 0x18 {
					entry, end, argc = 0x216, 0x22a, 2
				}
				for n := uint32(0); n < 64; n++ {
					h.mem[frame+n] = 0
				}
				h.poke(frame, uint16(parent-0x40000))
				for n := 1; n <= argc; n++ {
					h.poke(depotLocal(frame, n), h.peek(sp+uint32((n-1)*2)))
				}
				if proc == 0x17 {
					// 1個區域變數，五個參數因此從L2起。
					for n := argc; n >= 1; n-- {
						h.poke(depotLocal(frame, n+1), h.peek(depotLocal(frame, n)))
					}
					h.poke(depotLocal(frame, 1), 0)
				}
				levelsFrame(h, frame)
				foodRun(h, disk, entry, end, depth+1, calls)
				levelsFrame(h, parent)
				h.cpu.State.SSP = sp + uint32(argc*2)
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			if op == 0x73 && proc == 0x0a {
				frame := uint32(0x48000)
				for n := uint32(0); n < 64; n++ {
					h.mem[frame+n] = 0
				}
				for n := 1; n <= 3; n++ {
					h.poke(depotLocal(frame, n), h.peek(sp+uint32((n-1)*2)))
				}
				levelsFrame(h, frame)
				depotRun(h, disk, 0x2e800, 0x66c, 0x680, depth+1, nil)
				levelsFrame(h, parent)
				h.cpu.State.SSP = sp + 6
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			pop, handled := 0, false
			switch {
			case op == 0x73 && proc == 0x2f:
				if h.peek(sp) != 1 || h.peek(sp+2) != 0 {
					h.t.Fatal("食用前背景槽 ABI")
				}
				h.poke(sp+2, 60)
				pop, handled = 1, true
				*calls = append(*calls, [3]int{0x2f, 1, 60})
			case op == 0x73 && proc == 0x31:
				if h.peek(sp) != 1 {
					h.t.Fatal("食用停止背景槽 ABI")
				}
				pop, handled = 1, true
				*calls = append(*calls, [3]int{0x31, 1, 0})
			case op == 0x73 && proc == 0x30:
				if h.peek(sp) != 60 || h.peek(sp+2) != 1 {
					h.t.Fatal("食用恢復背景槽 ABI")
				}
				pop, handled = 2, true
				*calls = append(*calls, [3]int{0x30, 1, 60})
			case op == 0x91 && proc == 6:
				pop, handled = 1, true // 顯示字串。
			case op == 0x72 && proc == 0x20:
				pop, handled = 4, true // draw_icon。
			}
			if handled {
				h.cpu.State.SSP = sp + uint32(pop*2)
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("food native IP%x PC%x: %v", h.cpu.State.A[4], h.cpu.State.PC, err)
		}
	}
	h.t.Fatalf("food原始切片%x..%x超時", first, last)
}

// Stores selector6: 6+6-5=7, 28*7=196。兩個條件分支皆由原始TJP/FJP執行。
// WINDOWLI與MAINLIB全域分開；此框架只攔截繪圖與有界滑鼠/背景ABI。
func foodWindowGlobal(n int) uint32 { return 0x44008 + uint32(n*2) }
func foodChainHarness(t *testing.T, disk []byte) *harness {
	h := banvilleHarness(t, disk)
	h.poke(0x4100a, 0x2100)
	h.poke(0x42100, 0x4000)
	for n := uint32(0); n < 0x400; n++ {
		h.mem[0x44000+n] = 0
	}
	for _, seg := range [][3]int{{0x36600, 0xc00, 0x50000}, {0x31200, 0xc00, 0x52000}, {0x2e800, 0x2a00, 0x56000}} {
		for i, b := range disk[seg[0] : seg[0]+seg[1]] {
			h.mem[uint32(seg[2]+i)] = b
		}
	}
	h.poke(cadenceGlobal(8), 1)
	h.poke(0x49010, 24)
	h.poke(0x49012, 29)
	h.poke(0x49014, 0)
	for n := uint32(0); n < 12; n++ {
		h.mem[cadenceGlobal(0x773)+n] = 0
	}
	return h
}

func foodChainModule(h *harness, window bool) {
	base := uint32(0x45000)
	if window {
		base = 0x44000
	}
	h.cpu.State.A[1] = base
	h.poke(interpBase+0x1b70, uint16(base>>16))
	h.poke(interpBase+0x1b72, uint16(base))
}

// 原始完整Stores拖動：滑鼠座標預先固定到頭部或空白處；WINDOWLI:05真正hit-test。
func foodChainRun(h *harness, disk []byte, base, first, last, depth int, cancel bool, calls *[][3]int) {
	stream := uint32(operandStream + 0x7000 + depth*0x800)
	h.startDispatch(disk[base+first:base+last+4], stream)
	for steps := 0; steps < 100000; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp, parent := h.cpu.State.A[4], h.cpu.State.SSP, h.cpu.State.A[0]
			at := int(ip-stream) + first
			if at == last {
				return
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			childBase, entry, end, locals, argc, result := 0, 0, 0, 0, 0, 0
			switch {
			case op == 0x73 && proc == 7:
				childBase, entry, end, argc, result = 0x2e800, 0x5de, 0x5e2, 2, 3
			case op == 0x73 && proc == 0xa:
				childBase, entry, end, argc = 0x2e800, 0x66c, 0x680, 3
			case base == 0x36600 && op == 0xef && proc == 7:
				childBase, entry, end, locals, argc = base, 0x356, 0x3be, 3, 2
			case base == 0x36600 && op == 0xef && proc == 0xa:
				childBase, entry, end, locals, argc, result = base, 0x476, 0x4c0, 2, 2, 5
			case (op == 0x74 || base == 0x31200 && op == 0x91) && proc == 2:
				childBase, entry, end, locals, argc = 0x31200, 0x1a, 0x44, 1, 3
			case base == 0x31200 && op == 0x91 && proc == 0xa:
				childBase, entry, end, locals, argc = base, 0x3fe, 0x409, 1, 3
			case op == 0x74 && proc == 4:
				childBase, entry, end, locals, argc = 0x31200, 0x90, 0x122, 5, 5
			case op == 0x74 && proc == 5:
				childBase, entry, end, locals, argc = 0x31200, 0x128, 0x17d, 2, 4
			case op == 0x74 && proc == 8:
				childBase, entry, end, locals, argc = 0x31200, 0x2cc, 0x36e, 5, 4
			}
			if childBase != 0 {
				frame := uint32(0x4a000 + depth*0x100)
				for n := uint32(0); n < 100; n++ {
					h.mem[frame+n] = 0
				}
				link := parent
				if op == 0xef {
					link = 0x45000
				}
				h.poke(frame, uint16(link-0x40000))
				for n := 0; n < argc; n++ {
					h.poke(depotLocal(frame, locals+n+1), h.peek(sp+uint32(n*2)))
				}
				oldGlobal := h.cpu.State.A[1]
				oldSegment := h.cpu.State.A[2]
				cache0, cache1 := h.peek(interpBase+0x1b70), h.peek(interpBase+0x1b72)
				foodChainModule(h, childBase == 0x31200)
				switch childBase {
				case 0x31200:
					h.cpu.State.A[2] = 0x52000
				case 0x36600:
					h.cpu.State.A[2] = 0x50000
				case 0x2e800:
					h.cpu.State.A[2] = 0x56000
					h.cpu.State.A[1] = 0x43000
					h.poke(interpBase+0x1b72, 0x3000)
				}
				levelsFrame(h, frame)
				if childBase == 0x31200 && proc == 8 {
					foodRun(h, disk, entry, end, 0, calls)
				} else {
					foodChainRun(h, disk, childBase, entry, end, depth+1, cancel, calls)
				}
				if result != 0 {
					h.poke(sp+uint32(argc*2), h.peek(depotLocal(frame, result)))
				}
				levelsFrame(h, parent)
				h.cpu.State.A[1] = oldGlobal
				h.cpu.State.A[2] = oldSegment
				h.poke(interpBase+0x1b70, cache0)
				h.poke(interpBase+0x1b72, cache1)
				h.cpu.State.SSP = sp + uint32(argc*2)
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			pop, handled := 0, false
			switch {
			case op == 0x74 && proc == 3:
				pop, handled = 1, true // Stores視窗繪圖。
			case op == 0x74 && proc == 0x15:
				*calls = append(*calls, [3]int{int(h.peek(sp + 4)), int(h.peek(sp + 2)), int(h.peek(sp))})
				pop, handled = 3, true // Observe make_zoom arguments; rendering isolated.
			case op == 0x74 && (proc == 9 || proc == 0xb || proc == 6):
				pop, handled = 1, true
			case op == 0x91 && proc == 0xb && base == 0x31200:
				pop, handled = 1, true
			case op == 0x72 && proc == 0x20:
				pop, handled = 4, true
			case op == 0x72 && (proc == 0x27 || proc == 0x26):
				handled = true
			case op == 0x73 && proc == 2:
				handled = true // 輸入事件poll，不推進遊戲時鐘。
			case op == 0x73 && proc == 0x2a:
				h.poke(sp+2, 0)
				pop, handled = 1, true // 預先固定timeout false。
			case op == 0x73 && proc == 0x3c:
				for n, v := range []uint16{200, 0, 320, 0} {
					h.poke(0x40000+uint32(h.peek(sp+uint32(n*2))), v)
				}
				pop, handled = 5, true
			case op == 0x72 && proc == 0x1e:
				x, y := h.peek(0x49010), h.peek(0x49012)
				if cancel {
					x, y = 300, 180
				}
				h.poke(0x40000+uint32(h.peek(sp)), y)
				h.poke(0x40000+uint32(h.peek(sp+2)), x)
				pop, handled = 2, true
			case op == 0x72 && proc == 0x1f:
				if h.peek(sp) != 1 && h.peek(sp) != 0 {
					h.t.Fatal("滑鼠button ABI")
				}
				button := h.peek(sp)
				value := uint16(0)
				if button == 0 && !cancel || button == 1 && (cancel || h.peek(0x49014) == 1) {
					value = 1
				}
				h.poke(sp+2, value)
				pop, handled = 1, true // 執行前固定：食用(1,0)、取消(0,1)。
			}
			if handled {
				h.cpu.State.SSP = sp + uint32(pop*2)
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("chain base%x at%x PC%x: %v", base, int(h.cpu.State.A[4]-stream)+first, h.cpu.State.PC, err)
		}
	}
	h.t.Fatalf("chain base%x %x..%x超時 at%x PC%x frame%x held%d target%d item%d old%d head%d g3%d", base, first, last,
		int(h.cpu.State.A[4]-stream)+first, h.cpu.State.PC, h.cpu.State.A[0],
		h.peek(depotLocal(0x46000, 6)), h.peek(depotLocal(0x46000, 9)), h.peek(depotLocal(0x46000, 10)),
		h.peek(depotLocal(0x46000, 8)), h.peek(foodWindowGlobal(3)), h.peek(cadenceGlobal(3)))
}

func TestSunDogFoodStoresChain(t *testing.T) {
	disk := sundogWalkerDisk(t)
	type row struct {
		Slot, NourishmentBefore, NourishmentAfter, ItemAfter, HeldAfter int
		Cancel                                                          bool
		Flow                                                            string
		MiddleStores                                                    [20]byte
		MiddleCarry                                                     [4]byte
		StoresBefore, StoresAfter                                       [20]byte
		CarryBefore, CarryAfter                                         [4]byte
		Calls                                                           [][3]int
	}
	report := struct {
		Scope, Boundaries string
		Rows              []row
	}{
		Scope:      "XDOREPAI:03原始20格讀取→:0b完整拖動→:07持久清槽/取消還原→WINDOWLI:04槽交換/:05實際hit-test/:08頭部食用；兩個空攜帶槽先搬物再食用，已占用槽拒絕並還原；:0a原始適配檢查。MAINLIB:07原始位元AND/:0a原始座標夾限。",
		Boundaries: "明示MAINLIB/WINDOWLI獨立全域；XDOREPAI:02兩段原始選擇分支決定selector6→base196/unit7/direct-container=true，WINDOWLI:1d完整make_man/:0a讀取原始四攜帶槽。按鈕返回值食用(1,0)、取消(0,1)、搬物及占用槽(1,1)與目標座標於執行前固定；原版hit-test真正執行。圖形、游標、事件poll、timeout和背景槽採有界ABI；不宣稱整機開局/實體滑鼠/GUI/父時鐘。無RND。"}
	for slot := 0; slot < 20; slot++ {
		for _, n := range []int{0, 63, 64, 185, 186, 254, 255} {
			for _, flow := range []string{"eat", "cancel", "carry0", "carry3", "occupied1"} {
				cancel := flow == "cancel"
				h := foodChainHarness(t, disk)
				r := row{Slot: slot, NourishmentBefore: n, Cancel: cancel, Flow: flow}
				for i := range r.StoresBefore {
					r.StoresBefore[i] = byte(9 + i)
				}
				r.StoresBefore[slot] = 8
				r.CarryBefore = [4]byte{0, 11, 44, 0}
				for i, v := range r.StoresBefore {
					h.mem[cadenceGlobal(0x5b)+196+uint32(i)] = v
				}
				for i, v := range r.CarryBefore {
					h.mem[cadenceGlobal(0x3f)+uint32(i)] = v
				}
				before := [4]byte{113, 70, 169, byte(n)}
				for i, v := range before {
					h.mem[cadenceGlobal(0x3d)+uint32(i)] = v
				}
				h.mem[cadenceGlobal(0x35)] = 100
				foodChainModule(h, false)
				levelsFrame(h, 0x46000)
				h.poke(depotLocal(0x46000, 3), 6)
				foodChainRun(h, disk, 0x36600, 0x21, 0x2d, 0, cancel, &r.Calls)
				foodChainRun(h, disk, 0x36600, 0x10a, 0x139, 0, cancel, &r.Calls)
				if h.peek(depotLocal(0x45000, 1)) != 196 || h.peek(depotLocal(0x45000, 6)) != 7 || h.peek(depotLocal(0x45000, 0xb)) != 1 || h.peek(depotLocal(0x45000, 0xa)) != 0 {
					t.Fatal("原版Stores selector6分支不同")
				}
				foodChainModule(h, true)
				h.cpu.State.A[2] = 0x52000
				h.poke(depotLocal(0x46000, 1), 2)
				foodChainRun(h, disk, 0x31200, 0x876, 0x8ef, 0, cancel, &r.Calls)
				if h.peek(foodWindowGlobal(3)) != 3 || h.peek(foodWindowGlobal(4)) != 7 {
					t.Fatal("原版make_man攜帶槽不同")
				}
				foodChainModule(h, false)
				h.cpu.State.A[2] = 0x50000
				foodChainRun(h, disk, 0x36600, 0x140, 0x1ad, 0, cancel, &r.Calls)
				if h.peek(foodWindowGlobal(0xdc)) != 27 {
					t.Fatal("原版Stores非20格")
				}
				for i, v := range r.StoresBefore {
					if h.peek(foodWindowGlobal(0xdd)+uint32((i+7)*6+4)) != uint16(v) {
						t.Fatalf("原版Stores讀取槽%d不同", i)
					}
				}
				for i := uint32(0); i < 100; i++ {
					h.mem[0x46000+i] = 0
				}
				h.poke(0x46000, 0x5000)
				h.poke(depotLocal(0x46000, 15), uint16(slot+8))
				pocket := -1
				switch flow {
				case "carry0":
					pocket = 0
				case "carry3":
					pocket = 3
				case "occupied1":
					pocket = 1
				}
				if pocket >= 0 {
					coords := [][2]uint16{{27, 57}, {9, 60}, {9, 77}, {27, 77}}
					h.poke(0x49010, coords[pocket][0])
					h.poke(0x49012, coords[pocket][1])
					h.poke(0x49014, 1)
				}
				foodChainRun(h, disk, 0x36600, 0x4c6, 0x5dd, 0, cancel, &r.Calls)
				var middleStores [20]byte
				var middleCarry [4]byte
				for i := range middleStores {
					middleStores[i] = h.mem[cadenceGlobal(0x5b)+196+uint32(i)]
				}
				for i := range middleCarry {
					middleCarry[i] = h.mem[cadenceGlobal(0x3f)+uint32(i)]
				}
				r.MiddleStores, r.MiddleCarry = middleStores, middleCarry
				if pocket >= 0 && flow != "occupied1" {
					wantCarry := r.CarryBefore
					wantCarry[pocket] = 8
					wantStores := r.StoresBefore
					wantStores[slot] = 0
					if middleCarry != wantCarry || middleStores != wantStores || h.mem[cadenceGlobal(0x3d)+3] != byte(n) {
						t.Fatalf("原版搬到攜帶槽 flow%s %+v", flow, r)
					}
					h.poke(0x49010, 24)
					h.poke(0x49012, 29)
					h.poke(0x49014, 0)
					for i := uint32(0); i < 100; i++ {
						h.mem[0x46000+i] = 0
					}
					h.poke(0x46000, 0x5000)
					h.poke(depotLocal(0x46000, 15), uint16(pocket+4))
					foodChainRun(h, disk, 0x36600, 0x4c6, 0x5dd, 0, false, &r.Calls)
				}
				for i := range r.StoresAfter {
					r.StoresAfter[i] = h.mem[cadenceGlobal(0x5b)+196+uint32(i)]
				}
				for i := range r.CarryAfter {
					r.CarryAfter[i] = h.mem[cadenceGlobal(0x3f)+uint32(i)]
				}
				r.NourishmentAfter = int(h.mem[cadenceGlobal(0x3d)+3])
				r.HeldAfter = int(h.peek(depotLocal(0x46000, 6)))
				r.ItemAfter = int(h.peek(foodWindowGlobal(0xdd) + uint32((slot+7)*6+4)))
				wantStores, wantN := r.StoresBefore, n
				if !cancel && flow != "occupied1" {
					wantStores[slot] = 0
					wantN = min(255, n+70)
				}
				if r.StoresAfter != wantStores || r.CarryAfter != r.CarryBefore || r.NourishmentAfter != wantN || r.HeldAfter != 0 || r.ItemAfter != int(wantStores[slot]) {
					t.Fatalf("原始食物鏈 %+v", r)
				}
				for i := 0; i < 3; i++ {
					if h.mem[cadenceGlobal(0x3d)+uint32(i)] != before[i] {
						t.Fatal("非營養讀數改變")
					}
				}
				report.Rows = append(report.Rows, r)
			}
		}
	}
	data, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "native-food-stores-chain.json"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("原版Stores取物/頭部食用/取消還原%d案", len(report.Rows))
}

func TestSunDogFoodStoresWindow(t *testing.T) {
	disk := sundogWalkerDisk(t)
	h := foodChainHarness(t, disk)
	foodChainModule(h, false)
	h.cpu.State.A[2] = 0x50000
	levelsFrame(h, 0x46000)
	h.poke(depotLocal(0x46000, 3), 6)
	var calls [][3]int
	foodChainRun(h, disk, 0x36600, 0x46, 0x6f, 0, false, &calls)
	if len(calls) != 1 || calls[0] != ([3]int{3, 10, 0}) {
		t.Fatalf("Stores window %+v", calls)
	}
	report := struct {
		Calls [][3]int
		Scope string
	}{calls, "XDOREPAI:02 0046..006f原始XJP執行；make_zoom繪圖ABI只觀測參數(style,column,row)，不宣稱畫面或完整GUI。"}
	data, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "native-food-stores-window.json"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)

func foodConditionRun(h *harness, disk []byte, first, last, depth int) {
	stream := uint32(operandStream + 0xf000 + depth*0x400)
	h.startDispatch(disk[0x2e800+first:0x2e800+last+4], stream)
	for steps := 0; steps < 100000; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp, parent := h.cpu.State.A[4], h.cpu.State.SSP, h.cpu.State.A[0]
			if int(ip-stream)+first == last {
				return
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			if op == 0x90 && proc == 0x52 {
				frame := uint32(0x46200)
				for n := uint32(0); n < 64; n++ {
					h.mem[frame+n] = 0
				}
				h.poke(frame, uint16(parent-0x40000))
				for n := 4; n <= 6; n++ {
					h.poke(depotLocal(frame, n), h.peek(sp+uint32((n-4)*2)))
				}
				levelsFrame(h, frame)
				foodConditionRun(h, disk, 0xb62, 0xbf1, depth+1)
				levelsFrame(h, parent)
				h.cpu.State.SSP = sp + 6
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			if op == 0x91 && proc == 8 {
				frame := uint32(0x46400)
				h.poke(depotLocal(frame, 1), h.peek(sp))
				h.poke(depotLocal(frame, 2), h.peek(sp+2))
				levelsFrame(h, frame)
				foodConditionRun(h, disk, 0x5c4, 0x5d8, depth+1)
				levelsFrame(h, parent)
				h.cpu.State.SSP = sp + 2
				h.poke(sp+2, h.peek(depotLocal(frame, 2)))
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("condition depth%d at%x: %v", depth, int(h.cpu.State.A[4]-stream)+first, err)
		}
	}
	h.t.Fatal("condition超時")
}

func foodSleepMinute(h *harness, disk []byte, period int) int {
	h.poke(depotLocal(0x46000, 1), uint16(period))
	h.poke(depotLocal(0x46000, 2), 0xffff)
	stream := uint32(operandStream + 0xb000)
	h.startDispatch(disk[0x34800+0x5f3:0x34800+0x66a], stream)
	minutes := 0
	for steps := 0; steps < 100000; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp := h.cpu.State.A[4], h.cpu.State.SSP
			at := int(ip-stream) + 0x5f3
			if at == 0x666 {
				return minutes
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			if op == 0xef && proc == 4 && at == 0x625 {
				if h.peek(sp) != 0 {
					h.t.Fatal("睡眠等待helper參數")
				}
				h.cpu.State.SSP = sp + 2
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			if op == 0x73 && proc == 0x35 && at == 0x62b {
				if h.peek(sp) != 1 {
					h.t.Fatal("睡眠the_time分鐘參數")
				}
				minutes++
				h.cpu.State.SSP = sp + 2
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("sleep minute IP%x: %v", h.cpu.State.A[4], err)
		}
	}
	h.t.Fatal("sleep minute超時")
	return 0
}

func foodSleepRun(h *harness, disk []byte, base, first, last, depth int, trace *foodSleepTrace) {
	stream := uint32(0x60000 + depth*0x2000)
	h.startDispatch(disk[base+first:base+last+4], stream)
	stepLimit := 4000000
	// A deliberately artificial zero-delay initial table exercises the
	// original 16-bit do-until wait wrap: 65536 pulses, about 4.8M steps.
	if trace.Background != nil {
		stepLimit = 6000000
	}
	for steps := 0; steps < stepLimit; steps++ {
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip, sp, parent := h.cpu.State.A[4], h.cpu.State.SSP, h.cpu.State.A[0]
			at := int(ip-stream) + first
			if at == last {
				return
			}
			if base == 0x2e200 && at == 0x26 && trace.Background != nil {
				if trace.Background.Passes == trace.Background.Limit {
					return
				}
				trace.Background.Passes++
			}
			if base == 0x34800 && at == 0x613 {
				trace.Cycles++
			}
			if base == 0x2e200 && at == 0x70 && trace.Background != nil && trace.Background.StartupLocation > 0 && !trace.Background.StartupInitialized {
				// Register slots while the worker is in its empty-table first
				// wait. The original countdown writers execute unchanged.
				trace.Background.StartupInitialized = true
				foodSleepRun(h, disk, base, 0x311, 0x316, depth+1, trace)
				if trace.Background.StartupLocation == 2 {
					foodSleepRun(h, disk, base, 0x1b4, 0x1b8, depth+1, trace)
				}
				levelsFrame(h, parent)
				h.cpu.State.SSP, h.cpu.State.A[4] = sp, ip
				h.start(interpLoop)
				continue
			}
			op, proc := h.mem[ip], h.mem[ip+1]
			childBase, entry, end, locals, argc, result := 0, 0, 0, 0, 0, 0
			switch {
			case base == 0x37200 && op == 0x71 && proc == 0x14:
				childBase, entry, end, locals, argc = 0x31e00, 0x19a, 0x1b3, 1, 1
			case base == 0x37200 && op == 0x71 && proc == 0x13:
				childBase, entry, end, locals, argc = 0x31e00, 0x128, 0x15a, 2, 2
			case op == 0x73 && proc == 0x28 && trace.GroundClock != nil:
				childBase, entry, end, locals, result = 0x2e800, 0x8e4, 0x8f7, 2, 3

	disk := sundogWalkerDisk(t)

		Current, Target, After               [5]int
		MinuteBefore, MinuteAfter, HourAfter int
	}
	report := struct {
		Scope, Boundaries     string
		InitBefore, InitAfter [15]int
		ReadingsAfter         [4]int
		Rows                  []row
	}{Scope: "XSTARTUP raw0x39c00 :30 1bd1..1bf1速率初始化；MAINLIB raw0x2e800 :35 0bf8..0d20原始整點屬性恢復。",
		Boundaries: "從已確認的程序內部邊界進入，MAINLIB全body執行；圖形／lock隔離，無RND。非正常原版開局或GUI。"}
	h := foodSleepHarness(t, disk)
	for n := range report.InitBefore {
		report.InitBefore[n] = 17 + n
		h.mem[cadenceGlobal(0x35)+uint32(n)] = byte(report.InitBefore[n])
	}
	foodSleepRun(h, disk, 0x39c00, 0x1bd1, 0x1bf1, 0, &foodSleepTrace{})
	for n := range report.InitAfter {
		report.InitAfter[n] = int(h.mem[cadenceGlobal(0x35)+uint32(n)])
	}
	for n := range report.ReadingsAfter {
		report.ReadingsAfter[n] = int(h.mem[cadenceGlobal(0x3d)+uint32(n)])
	}
	for n := 0; n < 10; n++ {
		if report.InitAfter[n] != report.InitBefore[n] {
			t.Fatal("初始化改寫現值／目標")
		}
	}
	if report.InitAfter[10] != 5 || report.InitAfter[11] != 5 || report.InitAfter[12] != 5 || report.InitAfter[13] != 5 || report.InitAfter[14] != 1 || report.ReadingsAfter != [4]int{255, 255, 255, 255} {
		t.Fatalf("原始速率初始化 %v", report)
	}
	for _, current := range []int{0, 1, 4, 5, 6, 127, 254, 255} {
		for _, target := range []int{0, 1, 4, 5, 6, 127, 254, 255} {
			for _, minute := range []int{0, 59} {
				h := foodSleepHarness(t, disk)
				r := row{MinuteBefore: minute}
				for n := 0; n < 5; n++ {
					r.Current[n], r.Target[n] = current, target
					h.mem[cadenceGlobal(0x35)+uint32(n)] = byte(current)
					h.mem[cadenceGlobal(0x35)+uint32(n+5)] = byte(target)
					h.mem[cadenceGlobal(0x35)+uint32(n+10)] = byte(report.InitAfter[n+10])
				}
				for n := 0; n < 4; n++ {
					h.mem[cadenceGlobal(0x3d)+uint32(n)] = 255
				}
				h.poke(cadenceGlobal(0x11), uint16(minute))
				h.cpu.State.A[1], h.cpu.State.A[2] = 0x43000, 0x56000
				h.poke(interpBase+0x1b70, 4)
				h.poke(interpBase+0x1b72, 0x3000)
				h.poke(depotLocal(0x46000, 17), 1)
				foodSleepRun(h, disk, 0x2e800, 0xbf8, 0xd20, 0, &foodSleepTrace{})
				for n := range r.After {
					r.After[n] = int(h.mem[cadenceGlobal(0x35)+uint32(n)])
				}
				r.MinuteAfter, r.HourAfter = int(h.peek(cadenceGlobal(0x11))), int(h.peek(cadenceGlobal(0x12)))
				if h.cpu.State.SSP != stackTop {
					t.Fatal("屬性時鐘返回ABI")
				}
				report.Rows = append(report.Rows, r)
			}
		}
	}
	data, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "native-food-attribute-clock.json"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("原始恢復速率初始化與屬性整點／未整點：%d案", len(report.Rows))
}

// Load the original PRG at a synthetic base with its real relocation table.
// The bootstrap's BIOS pointer is text+4, after the program entry branch.
func foodPlatformClockHarness(t *testing.T, disk []byte) (*harness, uint32) {
	const prgBase = uint32(0x70000)
	raw := disk[0x2400 : 0x2400+3976]
	if fmt.Sprintf("%x", sha256.Sum256(raw)) != "b317a490ef64d110c535da69ee8bf164ed16e70eb9a7b0e5d77291cb33b7b902" {
		t.Fatal("原始PRG身分")
	}
	code := append([]byte(nil), raw[28:28+0xf3c]...)
	relocs := raw[28+0xf3c:]
	at := int(binary.BigEndian.Uint32(relocs[:4]))
	binary.BigEndian.PutUint32(code[at:at+4], binary.BigEndian.Uint32(code[at:at+4])+prgBase)
	for _, delta := range relocs[4:] {
		if delta == 0 {
			break
		}
		if delta == 1 {
			at += 254
			continue
		}
		at += int(delta)
		if at < 0 || at+4 > len(code) {
			t.Fatal("PRG relocation bounds")
		}
		binary.BigEndian.PutUint32(code[at:at+4], binary.BigEndian.Uint32(code[at:at+4])+prgBase)
	}
	if binary.BigEndian.Uint32(code[0x334:0x338]) != prgBase+4 {
		t.Fatal("原始bootstrap BIOS pointer")
	}
	h := newHarness(t, loadInterp(t))
	for n, b := range code {
		h.mem[prgBase+uint32(n)] = b
	}
	h.cpu.State.A[6] = 0x40000
	pokeLong := func(a, v uint32) { h.poke(a, uint16(v>>16)); h.poke(a+2, uint16(v)) }
	pokeLong(interpBase+0x1c28, prgBase+4)
	pokeLong(interpBase+0x1c30, 0x60000)
	return h, prgBase
}

func TestSunDogFoodPlatformClock(t *testing.T) {
	disk := sundogWalkerDisk(t)
	type row struct {
		Before, After           uint32

	}

			for _, vbls := range []int{1, 4, 9} {

				pokeLong := func(a, v uint32) { h.poke(a, uint16(v>>16)); h.poke(a+2, uint16(v)) }
				pokeLong(base+0x18c, before)

				h.push(uint16(sentinel >> 16))
				entry := base + 4 + 0x70
				if enabled {
					entry = base + 4 + 0x74
				}
				h.start(entry - interpBase)
				h.run(1000)
				readTime := func() [2]int {
					h.poke(0x4f000, 0x1234)
					h.poke(0x4f002, 0x5678)
					h.push(0xf000)
					h.push(0xf002)
					h.start(0x1832)
					h.run(1000)
					if h.cpu.State.SSP != stackTop {
						t.Fatal("原始TIME stack")
					}
					return [2]int{int(h.peek(0x4f000)), int(h.peek(0x4f002))}
				}

							continue
						}
						if _, err := h.cpu.Step(); err != nil {
							t.Fatal("原始VBL prefix", err)
						}
					}
					if !done || h.cpu.State.SSP != stackTop {
						t.Fatal("VBL prefix bounds")
					}
				}
				r.After = uint32(h.peek(base+0x18c))<<16 | uint32(h.peek(base+0x18e))
				r.Countdown = int(h.peek(base + 0x190))
				r.TimeAfter = readTime()
				rows = append(rows, r)
			}
		}
	}
	report := struct {
		Scope, Boundaries string
		Rows              []row
	}{
		Scope:      "原始PRG重定位與bootstrap BIOS base+4；原始QUIET/ENABLE、VBL計數前綴與SYSTEM.INTERP KERNEL:14 TIME及BIOS真正執行。",
		Boundaries: "固定初態36案；人工送入VBL前綴，不執行舊TOS handler、rte、完整IRQ/signal或GUI。原始event0 callback只記ABI後停止；不比較每秒Hz或硬體wall-clock，沒有RND。", Rows: rows}

							for n := range r.StoresAfter {
								r.StoresAfter[n] = h.mem[cadenceGlobal(0x5b)+196+uint32(n)]
							}
							for n := range r.After {
								r.After[n] = h.mem[cadenceGlobal(0x3d)+uint32(n)]
							}
							gateEnabled := r.Bit13 && !r.Flag57 && (r.LocationMode == 3 || r.LocationMode == 5)
							if r.Trace.Minutes != minutes || r.Trace.Polls != minutes || h.cpu.State.SSP != stackTop || r.StoresAfter != r.StoresBefore {
								t.Fatalf("失竊鏈時間／返回／Stores旁效 %v SSP%x", r, h.cpu.State.SSP)
							}
							if (!gateEnabled || risk >= minutes) && (len(r.Trace.Random) != 0 || r.CashAfter != cash || r.CarryAfter != carry || r.SeedAfter != seed) {
								t.Fatalf("失竊關閉gate未保持 %v", r)
							}
							if gateEnabled && risk < minutes && len(r.Trace.Random) == 0 {
								t.Fatalf("失竊開啟gate未使用原始RND %v", r)
							}
							if r.CashAfter != 0 && r.CashAfter != cash {
								t.Fatalf("原始失竊現金writer %v", r)
							}
							for n, v := range r.CarryAfter {
								if v != 0 && v != carry[n] {
									t.Fatalf("原始失竊固定攜帶槽writer %v", r)
								}
							}
							report.Rows = append(report.Rows, r)
						}
					}
				}
			}
		}
	}
	data, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "native-food-sleep-theft.json"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	t.Logf("完整睡眠失竊與固定原始種子：%d案", len(report.Rows))
}
