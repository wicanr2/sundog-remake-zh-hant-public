package main

import (
	"crypto/sha256"
	"encoding/json"
	"fmt"
	"image"
	"image/png"
	"os"
	"path/filepath"
	"reflect"
	"testing"

	"github.com/wicanr2/sundog-remake/internal/game"
	"github.com/wicanr2/sundog-remake/internal/localization"
	"github.com/wicanr2/sundog-remake/internal/source/atarist/originalmedia"
)

func TestHDStoresExport(t *testing.T) {
	a, _ := hdMapsTestApp(t)
	source := os.Getenv("HD_STORES_SOURCE")
	raw, err := os.ReadFile(source)
	if err != nil {
		t.Fatal(err)
	}
	var wrapper struct {
		State game.State `json:"state"`
	}
	if err = json.Unmarshal(raw, &wrapper); err != nil {
		t.Fatal(err)
	}
	if wrapper.State.Screen != game.ScreenShipboard {
		t.Fatal("source not shipboard")
	}
	state, err := a.reducer.Step(wrapper.State, game.ActionOpenStores)
	if err != nil {
		t.Fatal(err)
	}
	if state.Screen != game.ScreenStores {
		t.Fatal("normal rule entry")
	}
	diskPath := os.Getenv("HD_WORLD_DISK")
	diskRaw, err := os.ReadFile(diskPath)
	if err != nil {
		t.Fatal(err)
	}
	disk, err := originalmedia.Open(diskPath)
	if err != nil {
		t.Fatal(err)
	}
	picture, err := disk.DecodeImage(4)
	if err != nil {
		t.Fatal(err)
	}
	if picture.Width != 496 || picture.Height != 16 {
		t.Fatal("image4 shape")
	}
	var expected struct {
		Icons []struct {
			PixelSHA256 string `json:"pixelSha256"`
		}
	}
	b, err := os.ReadFile("docs/atari-st/item-icons.json")
	if err != nil {
		t.Fatal(err)
	}
	if err = json.Unmarshal(b, &expected); err != nil {
		t.Fatal(err)
	}
	if len(expected.Icons) != 31 {
		t.Fatal("source icon count")
	}
	var icons []map[string]any
	for i := 0; i < 31; i++ {
		pixels := make([]byte, 0, 256)
		for y := 0; y < 16; y++ {
			pixels = append(pixels, picture.Pixels[y*496+i*16:y*496+(i+1)*16]...)
		}
		h := fmt.Sprintf("%x", sha256.Sum256(pixels))
		if h != expected.Icons[i].PixelSHA256 {
			t.Fatal("original icon hash", i)
		}
		name, ok := game.ItemName(i + 1)
		if !ok {
			t.Fatal("item name", i)
		}
		icons = append(icons, map[string]any{"code": i + 1, "sourceIndex": i, "sourcePixelSHA256": h, "name": name})
	}
	out := os.Getenv("HD_STORES_OUTPUT")
	if out == "" {
		t.Fatal("output required")
	}
	palette := originalmedia.PaletteRGBA()
	original := image.NewRGBA(image.Rect(0, 0, 496, 16))
	for y := 0; y < 16; y++ {
		for x := 0; x < 496; x++ {
			original.SetRGBA(x, y, palette[picture.Pixels[y*496+x]&15])
		}
	}
	f, err := os.Create(filepath.Join(filepath.Dir(out), "source-icons-local.png"))
	if err != nil {
		t.Fatal(err)
	}
	if err = png.Encode(f, original); err != nil {
		t.Fatal(err)
	}
	if err = f.Close(); err != nil {
		t.Fatal(err)
	}
	window := stZoomFor(game.ScreenStores, 0, 0, 0)
	var slots []map[string]any
	appendSlot := func(kind string, index int, slot stSlot, item int) {
		x, y := stRepairSlotRect(window, slot)
		slots = append(slots, map[string]any{"kind": kind, "index": index, "x": x, "y": y, "w": 16, "h": 16, "cx": window.x0 + 1 + slot.x, "cy": window.y0 + 1 + slot.y, "item": item, "code": game.ItemCode(item)})
	}
	for i := 0; i < 20; i++ {
		appendSlot("store", i, storeGridSlot(i), state.Stores[i])
	}
	for i, slot := range stRepairPockets {
		appendSlot("pocket", i, slot, state.Parts[i])
	}
	appendSlot("hand", 0, stRepairHand, 0)
	for _, entry := range slots {
		x, y := entry["x"].(int), entry["y"].(int)
		for _, delta := range [][2]int{{0, 0}, {15, 0}, {0, 15}, {15, 15}, {8, 8}} {
			a.state = state
			action, hit := a.stStoresClick(x+delta[0], y+delta[1])
			if !hit {
				t.Fatal("slot hit", entry, delta)
			}
			expectedState := state
			switch entry["kind"] {
			case "store":
				expectedState = a.reducer.SelectStoreItem(state, entry["index"].(int))
			case "pocket":
				expectedState = a.reducer.PickPocketPart(state, entry["index"].(int))
			case "hand":
				if action != game.ActionEat {
					t.Fatal("head action")
				}
			}
			if !reflect.DeepEqual(a.state, expectedState) {
				t.Fatal("slot hit state", entry, delta)
			}
		}
	}
	var languages []map[string]any
	for _, locale := range []localization.Locale{"zh-Hant", "zh-Hans", "ja", "en"} {
		a.applyLocale(locale)
		languages = append(languages, map[string]any{"locale": locale, "title": a.lookup("ui.ship_stores"), "footer": a.lookup("ui.stores_short_keys")})
	}
	model := map[string]any{"schema": "sundog-hd-stores-fixture-1", "status": "DRAFT-geometry-export-not-normal-GUI", "sourceSave": source, "sourceSaveSHA256": fmt.Sprintf("%x", sha256.Sum256(raw)), "diskSHA256": fmt.Sprintf("%x", sha256.Sum256(diskRaw)), "sourceImage": 4, "sourceRawOffset": "0x11552", "icons": icons, "state": state, "window": []int{window.x0, window.y0, window.x1 - window.x0 + 1, window.y1 - window.y0 + 1}, "slots": slots, "manOutline": stManOutline, "partsBox": []int{window.x0 + 1 + stRepairPartsBox.x0, window.y0 + 1 + stRepairPartsBox.y0, stRepairPartsBox.x1 - stRepairPartsBox.x0 + 1, stRepairPartsBox.y1 - stRepairPartsBox.y0 + 1}, "strip": []int{window.x0 + 1 + stRepairStrip.x0, window.y0 + 1 + stRepairStrip.y0, stRepairStrip.x1 - stRepairStrip.x0 + 1, stRepairStrip.y1 - stRepairStrip.y0 + 1}, "languages": languages, "titleSafe": []int{window.x0 + stZoomInner, window.y0 + stZoomTitleTextY, stZoomW - 2*stZoomInner, stZoomTitleH - 2}, "footerSafe": []int{window.x0 + 3, window.y0 + 89, stZoomW - 6, 8}, "hitChecks": 125, "limits": "Fixture from a real shipboard source via normal reducer action. Source image pixels exported only locally. Original GUI parity and new art approval are separate."}
	data, err := json.MarshalIndent(model, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(out, append(data, '\n'), 0644); err != nil {
		t.Fatal(err)
	}
	t.Log("PASS original31 icon hashes;25 slots x5 hit/state checks;4 locales;formal geometry exported")
}
