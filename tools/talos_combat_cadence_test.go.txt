// 僅研究測試；非正式規則。定位、原始輸入與限制見 talos-combat-audit-20260907.md。
type cadenceInput struct {
	Enemy, Mode, Timer, Range, Phase, Drift, Seed int
	Flag44, Flag12, Ready                         bool
}
type cadenceResult struct {
	Input                          cadenceInput
	Timer                          int
	Ready, Flag69                  bool
	Counter                        int
	Draws                          []combatDraw
	ConsumedTimer, ConsumedCounter int
	ConsumedDraws                  []combatDraw
}

type cadenceCourseResult struct {
	Timer, Display, Range, Closing                   int
	TimerAfter, RangeAfter, BearingAfter, PitchAfter int
	Refresh                                          bool
}

func cadenceHarness(t *testing.T, disk []byte) *harness {
	h := newHarness(t, loadInterp(t))
	h.cpu.State.A[6], h.cpu.State.A[1], h.cpu.State.A[0], h.cpu.State.A[2] = 0x40000, 0x45000, 0x46000, 0x50000
	h.poke(interpBase+0x1b8a, 0x1000)
	h.poke(0x41008, 0x2000)
	h.poke(0x42000, 0x3000)
	for at := uint32(stackTop - 512); at < stackTop+64; at++ {
		h.mem[at] = 0
	}
	for _, bounds := range [][2]uint32{{0x43000, 0x43400}, {0x45000, 0x46100}, {0x47200, 0x47280}, {0x48000, 0x48100}, {0x49000, 0x49100}} {
		for at := bounds[0]; at < bounds[1]; at++ {
			h.mem[at] = 0
		}
	}
	// IDA base 0：XJP 0xd64..0xda0、讀字 0xd58..0xd64。
	// 原始標頭 byte 0xc=0，常數池 offset 0x0965；原始跳表不重建、不代算分派。
	for i, b := range disk[0x31e00:0x31e20] {
		h.mem[0x50000+uint32(i)] = b
	}
	for i, b := range disk[0x330dc:0x33124] {
		h.mem[0x512dc+uint32(i)] = b
	}
	h.poke(interpBase+0x1b6c, 4)
	h.poke(interpBase+0x1b6e, 0x6000) // XJP 返回時還原區域 frame A0。
	for i, b := range disk[0x4000:0x401c] {
		h.mem[0x47200+uint32(i)] = b
	}
	h.poke(cadenceOuter(6), 0x7000)
	return h
}
func cadenceGlobal(n int) uint32 { return 0x43008 + uint32(n*2) }
func cadenceOuter(n int) uint32  { return 0x45008 + uint32(n*2) }
func cadenceFlag(h *harness, n int, value bool) {
	a := cadenceGlobal(0x53) + uint32(n/16*2)
	v := h.peek(a) &^ (uint16(1) << uint(n%16))
	if value {
		v |= uint16(1) << uint(n%16)
	}
	h.poke(a, v)
}
func cadenceFlagValue(h *harness, n int) bool {
	return h.peek(cadenceGlobal(0x53)+uint32(n/16*2))&(uint16(1)<<uint(n%16)) != 0
}
func cadenceDraw(counter *int, low, high int, draws *[]combatDraw) int {
	(*counter)++
	mixed := uint32(*counter)*1664525 + 1013904223
	mixed ^= mixed >> 16
	v := low + int(mixed%uint32(high-low+1))
	*draws = append(*draws, combatDraw{low, high, v, *counter})
	return v
}
func cadenceRun(h *harness, disk []byte, first, last int, counter *int, draws *[]combatDraw) {
	cadenceRunAt(h, disk, first, last, operandStream, counter, draws)
}
func cadenceRunAt(h *harness, disk []byte, first, last int, stream uint32, counter *int, draws *[]combatDraw) {
	cadenceRunAtStops(h, disk, first, last, stream, []int{last}, counter, draws)
}

// 多出口只用於原版攻擊前綴；停止點由原版IP決定，不代算ready分支。
func cadenceRunAtStops(h *harness, disk []byte, first, last int, stream uint32, stops []int, counter *int, draws *[]combatDraw) {
	h.startDispatch(disk[0x31e00+first:0x31e00+last+4], stream)
	for steps := 0; ; steps++ {
		if steps >= 10000 {
			h.t.Fatal("排程切片超出步數")
		}
		if h.cpu.State.PC == interpBase+interpLoop+4 {
			ip := h.cpu.State.A[4]
			for _, stop := range stops {
				if ip == stream+uint32(stop-first) {
					return
				}
			}
			if h.mem[ip] == 0x91 {
				if first == 0xd23 && (h.mem[ip+1] == 0x1e || h.mem[ip+1] == 0x20) {
					oldA0, oldSP := h.cpu.State.A[0], h.cpu.State.SSP
					calleeFirst, calleeLast, calleeStream := 0x650, 0x8e4, uint32(operandStream+0x1000)
					if h.mem[ip+1] == 0x20 {
						calleeFirst, calleeLast, calleeStream = 0xc50, 0xcee, operandStream+0x1400
					}
					cadenceRunAt(h, disk, calleeFirst, calleeLast, calleeStream, counter, draws)
					h.cpu.State.A[0], h.cpu.State.SSP = oldA0, oldSP
					h.cpu.State.A[4] = ip + 2
					h.start(interpLoop)
					continue
				}
				if first == 0xb9e && (h.mem[ip+1] == 0x0b || h.mem[ip+1] == 0x1e) {
					oldA0, oldSP := h.cpu.State.A[0], h.cpu.State.SSP
					calleeFirst, calleeLast := 0x52c, 0x599
					calleeStream := uint32(operandStream + 0x400)
					if h.mem[ip+1] == 0x1e {
						calleeFirst, calleeLast, calleeStream = 0x650, 0x8e4, operandStream+0x1000
					}
					cadenceRunAt(h, disk, calleeFirst, calleeLast, calleeStream, counter, draws)
					h.cpu.State.A[0], h.cpu.State.SSP = oldA0, oldSP
					h.cpu.State.A[4] = ip + 2
					h.start(interpLoop)
					continue
				}
				if first != 0x650 || h.mem[ip+1] != 0x1d {
					h.t.Fatal("未知航向子程序")
				}
				oldA0, oldSP := h.cpu.State.A[0], h.cpu.State.SSP
				h.cpu.State.A[0] = 0x48000
				h.poke(interpBase+0x1b6c, 4)
				h.poke(interpBase+0x1b6e, 0x8000)
				cadenceRunAt(h, disk, 0x5a0, 0x64a, operandStream+0x800, counter, draws)
				h.cpu.State.A[0], h.cpu.State.SSP = oldA0, oldSP
				h.poke(interpBase+0x1b6c, uint16(oldA0>>16))
				h.poke(interpBase+0x1b6e, uint16(oldA0))
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
			if h.mem[ip] == 0x73 {
				sp := h.cpu.State.SSP
				switch h.mem[ip+1] {
				case 0x0b:
					low, high := 0, 1
					if h.mem[ip+2] == 0x80 && h.mem[ip+3] == 100 && h.mem[ip+4] == 0x8f {
						high = 99
					} else if h.mem[ip+2] >= 2 && h.mem[ip+2] <= 5 && h.mem[ip+3] == 0x8f {
						high = int(h.mem[ip+2]) - 1
					} else if h.mem[ip+2] == 0x81 && h.mem[ip+3] == 0 && h.mem[ip+4] == 4 && h.mem[ip+5] == 0x8f {
						high = 1023
					} else if h.mem[ip+2] != 0xa1 && h.mem[ip+2] != 0xd4 {
						h.t.Fatal("未知 random() consumer")
					}
					h.poke(sp, uint16(cadenceDraw(counter, low, high, draws)))
				case 0x0c:
					high, low := int(int16(h.peek(sp))), int(int16(h.peek(sp+2)))
					if high < low {
						h.t.Fatal("亂數界限反轉")
					}
					h.cpu.State.SSP += 4
					h.poke(h.cpu.State.SSP, uint16(cadenceDraw(counter, low, high, draws)))
				case 0x03:
					if h.peek(sp) != 26 {
						h.t.Fatal("未知聲音參數")
					}
					h.cpu.State.SSP += 2 // 只省略 pirate laser 聲音，不代算排程。
				case 0x0a:
					oldA0, oldSP := h.cpu.State.A[0], h.cpu.State.SSP
					high, ptr, low := h.peek(sp), h.peek(sp+2), h.peek(sp+4)
					h.cpu.State.A[0] = 0x49000
					h.poke(0x4900a, high)
					h.poke(0x4900c, ptr)
					h.poke(0x4900e, low)
					h.startDispatch(disk[0x2ee6c:0x2ee84], operandStream+0xc00)
					for n := 0; !(h.cpu.State.PC == interpBase+interpLoop+4 && h.cpu.State.A[4] == operandStream+0xc00+0x14); n++ {
						if n >= 2048 {
							h.t.Fatal("原版夾限未終止")
						}
						if _, err := h.cpu.Step(); err != nil {
							h.t.Fatal(err)
						}
					}
					h.cpu.State.A[0], h.cpu.State.SSP = oldA0, oldSP+6
				default:
					h.t.Fatalf("非預期呼叫 %x", h.mem[ip+1])
				}
				h.cpu.State.A[4] = ip + 2
				h.start(interpLoop)
				continue
			}
		}
		if _, err := h.cpu.Step(); err != nil {
			h.t.Fatalf("原生排程 PC=%x IP=%x: %v", h.cpu.State.PC-4, h.cpu.State.A[4], err)
		}
	}
}

func TestSunDogCombatCadence(t *testing.T) {
	disk, err := os.ReadFile(os.Getenv("SUNDOG_PARITY_DISK"))
	if err != nil {
		t.Fatal(err)
	}
	if fmt.Sprintf("%x", sha256.Sum256(disk)) != "f7c541b3cffdc7befc5641d667f039a8b33b0dfc406d17b20844158e52155fd6" {
		t.Fatal("磁碟身分不同")
	}
	inputs := []cadenceInput{}
	for enemy := 0; enemy < 7; enemy++ {
		for mode := 0; mode < 8; mode++ {
			for _, timer := range []int{-1, 1, 10} {
				for _, distance := range []int{39, 69, 79, 80, 100} {
					for phase := 0; phase < 3; phase++ {
						for _, seed := range []int{0, 4} {
							inputs = append(inputs, cadenceInput{enemy, mode, timer, distance, phase, []int{-4, -3, 3, 4}[(mode+phase+seed)%4], seed, true, false, false})
						}
					}
				}
			}
		}
	}
	for _, f44 := range []bool{false, true} {
		for _, f12 := range []bool{false, true} {
			for _, ready := range []bool{false, true} {
				for _, drift := range []int{-4, -3, 3, 4} {
					inputs = append(inputs, cadenceInput{0, 6, 1, 39, 0, drift, 0, f44, f12, ready})
				}
			}
		}
	}
	report := struct {
		Scope, Randomness                                                       string
		Cases, Differences, PendingCases                                        int
		CourseCases, CourseDifferences                                          int
		FormalProjectionCases, FormalTimerDifferences, FormalCounterDifferences int
		Native                                                                  []cadenceResult
		CourseNative                                                            []cadenceCourseResult
	}{Scope: "原版完整 set_fire c50..cee，含原始八模式 XJP、冷卻減法、旗標門檻、ready 與通知寫回；另驗 d86..da3 ready 消費／冷卻重設。非正式重製規則、非完整戰鬥／全機時間。", Randomness: "執行前固定 Seed，以同 counter 明示供應 random() 消費餘數／最低位及 rand 請求；非原版 RNG 對拍。"}
	var formal []struct {
		Input          cadenceInput
		Timer, Counter int
	}
	formalRaw, err := os.ReadFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "remake-cadence.json"))
	if err != nil {
		t.Fatal(err)
	}
	if err = json.Unmarshal(formalRaw, &formal); err != nil || len(formal) != 5040 {
		t.Fatal("正式投影矩陣不符")
	}
	for _, r := range inputs {
		h := cadenceHarness(t, disk)
		h.poke(cadenceOuter(10), uint16(r.Timer))
		if r.Ready {
			h.poke(cadenceOuter(5), 1)
		}
		for index, value := range map[int]int{0xd8: r.Enemy, 0xce: r.Mode, 0xdb: r.Range, 0x19: r.Phase, 0xd2: r.Drift} {
			h.poke(cadenceGlobal(index), uint16(value))
		}
		cadenceFlag(h, 44, r.Flag44)
		cadenceFlag(h, 12, r.Flag12)
		actual := cadenceResult{Input: r, Counter: r.Seed}
		cadenceRun(h, disk, 0xc50, 0xcee, &actual.Counter, &actual.Draws)
		actual.Timer = int(int16(h.peek(cadenceOuter(10))))
		actual.Ready = h.peek(cadenceOuter(5)) != 0
		actual.Flag69 = cadenceFlagValue(h, 69)
		// 獨立、僅測試的 typed 契約。不得直接接到 production 或保存格式。
		want := cadenceResult{Input: r, Timer: r.Timer, Ready: r.Ready, Counter: r.Seed}
		if want.Timer > 0 {
			want.Timer -= int(disk[0x4000+r.Enemy*4+3])
		}
		if want.Timer < 1 && r.Flag44 && !r.Flag12 {
			switch r.Mode {
			case 0:
				want.Ready = cadenceDraw(&want.Counter, 0, 99, &want.Draws) > r.Range
			case 2:
				value := cadenceDraw(&want.Counter, 0, 1, &want.Draws)
				want.Ready = r.Range < 80 && value != 0 && r.Phase == 0
			case 6:
				value := cadenceDraw(&want.Counter, 40, 70, &want.Draws)
				want.Ready = r.Range < value && (r.Drift > 3 || r.Drift < -3)
			default:
				want.Ready = r.Range < cadenceDraw(&want.Counter, 40, 80, &want.Draws)
			}
		}
		want.Flag69 = want.Ready
		report.Cases++
		report.Native = append(report.Native, actual)
		if report.Cases <= len(formal) {
			f := formal[report.Cases-1]
			if f.Input != r {
				t.Fatal("正式投影輸入不同")
			}
			report.FormalProjectionCases++
			if f.Timer != actual.Timer {
				report.FormalTimerDifferences++
			}
			if f.Counter != actual.Counter {
				report.FormalCounterDifferences++
			}
		}
		if !reflect.DeepEqual(actual, want) {
			report.Differences++
			if report.Differences < 6 {
				t.Logf("排程不同：native=%+v contract=%+v", actual, want)
			}
		}
		if actual.Ready {
			// 原版 d86 分支進入下一拍射擊前綴，而不是再次呼叫 set_fire。
			resetCounter := actual.Counter
			draws := []combatDraw{}
			cadenceRun(h, disk, 0xd86, 0xda3, &resetCounter, &draws)
			if len(draws) != 1 || draws[0].Low != 5 || draws[0].High != 10 || int(h.peek(cadenceOuter(10))) != draws[0].Value || h.peek(cadenceOuter(5)) != 0 || !cadenceFlagValue(h, 43) || resetCounter != actual.Counter+1 {
				t.Fatal("ready 消費／冷卻重設不符")
			}
			report.Native[report.Cases-1].ConsumedTimer = int(h.peek(cadenceOuter(10)))
			report.Native[report.Cases-1].ConsumedCounter = resetCounter
			report.Native[report.Cases-1].ConsumedDraws = draws
			report.PendingCases++
		}
	}
	for _, timer := range []int{0, 1, 2, 10} {
		for _, display := range []int{0, 1, 2} {
			for _, distance := range []int{9, 10, 128} {
				for _, closing := range []int{-3, 0, 3} {
					h := cadenceHarness(t, disk)
					// IDA 0x8c2..0x8d6：LDP 先 pop shift=8，再 pop width=4。
					for n, v := range map[int]int{0xd9: timer, 0xdb: distance, 0xd1: closing, 0xd0: 512, 0xcf: 512, 0xd2: -4, 0xd3: 3, 0x32: display << 8} {
						h.poke(cadenceGlobal(n), uint16(v))
					}
					h.startDispatch(disk[0x32b23:0x32b8a], operandStream)
					refresh := false
					for steps := 0; ; steps++ {
						if steps >= 6000 {
							t.Fatal("航向選擇切片未終止")
						}
						if h.cpu.State.PC == interpBase+interpLoop+4 {
							ip := h.cpu.State.A[4]
							if ip == operandStream+0xd84-0xd23 {
								refresh = true
								break
							}
							if ip == operandStream+0xd86-0xd23 {
								break
							}
						}
						if _, err := h.cpu.Step(); err != nil {
							t.Fatal(err)
						}
					}
					wantTimer, wantRange, wantBearing, wantPitch := timer, distance, 512, 512
					if timer > 0 {
						wantTimer--
						wantRange = distance + closing
						if wantRange < 10 {
							wantRange = 10
						}
						if display != 1 {
							wantBearing -= 12
							wantPitch -= 16
						}
					}
					report.CourseCases++
					report.CourseNative = append(report.CourseNative, cadenceCourseResult{timer, display, distance, closing, int(h.peek(cadenceGlobal(0xd9))), int(h.peek(cadenceGlobal(0xdb))), int(h.peek(cadenceGlobal(0xd0))), int(h.peek(cadenceGlobal(0xcf))), refresh})
					if refresh != (timer <= 0) || int(h.peek(cadenceGlobal(0xd9))) != wantTimer || int(h.peek(cadenceGlobal(0xdb))) != wantRange || int(h.peek(cadenceGlobal(0xd0))) != wantBearing || int(h.peek(cadenceGlobal(0xcf))) != wantPitch {
						report.CourseDifferences++
						if report.CourseDifferences < 7 {
							t.Logf("航向 timer=%d display=%d range=%d closing=%d refresh=%v native=(%d,%d,%d,%d) want=(%d,%d,%d,%d)", timer, display, distance, closing, refresh, h.peek(cadenceGlobal(0xd9)), h.peek(cadenceGlobal(0xdb)), h.peek(cadenceGlobal(0xd0)), h.peek(cadenceGlobal(0xcf)), wantTimer, wantRange, wantBearing, wantPitch)
						}
					}
				}
			}
		}
	}
	data, err := json.MarshalIndent(report, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(filepath.Join(os.Getenv("SUNDOG_PARITY_OUTPUT"), "native-cadence.json"), append(data, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
	if report.Differences != 0 || report.CourseDifferences != 0 {
		t.Fatalf("排程 %d/%d、航向 %d/%d 契約差異", report.Differences, report.Cases, report.CourseDifferences, report.CourseCases)
	}
	t.Logf("原生 set_fire %d 案、ready 消費 %d 案通過；正式規則尚未更動", report.Cases, report.PendingCases)
}
